 3553 | create or replace function public.aplicar_matriculas(p_persona uuid)
 3554 | returns int
 3555 | language plpgsql
 3556 | security definer
 3557 | set search_path = public
 3558 | as $$
 3559 | declare
 3560 |   v_correo text;
 3561 |   v_actual text;
 3562 |   v_plan   text;
 3563 |   v_suya   uuid;
 3564 |   v_inst   uuid;
 3565 |   v_rol    text;
 3566 |   v_cabe   boolean;
 3567 |   v_hechas int := 0;
 3568 | begin
 3569 |   select correo, rol, plan, institucion_id
 3570 |     into v_correo, v_actual, v_plan, v_suya
 3571 |     from public.perfiles where id = p_persona;
 3572 |   if v_correo is null then return 0; end if;
 3573 | 
 3574 |   -- De qué institución es esta persona.
 3575 |   --
 3576 |   -- Si ya pertenece a una, solo se le aplica lo de ESA. La nómina de otra no
 3577 |   -- se lleva a alguien que ya está en un contrato: sin esta regla, la
 3578 |   -- secretaría de la universidad B podría escribir el correo de un alumno de
 3579 |   -- la A en su planilla y quedárselo, con sus ramos y todo.
 3580 |   if v_suya is not null then
 3581 |     v_inst := v_suya;
 3582 |   else
 3583 |     -- Si no, la primera que lo pidió.
 3584 |     select institucion_id into v_inst
 3585 |       from public.matriculas
 3586 |      where correo = v_correo and aplicada_en is null
 3587 |      order by creada_en
 3588 |      limit 1;
 3589 |   end if;
 3590 |   if v_inst is null then return 0; end if;
 3591 | 
 3592 |   -- El rol más alto que le asigna la nómina de esa institución. Una persona
 3593 |   -- puede aparecer muchas veces —una por ramo— y todas dicen lo mismo, pero
 3594 |   -- si no, el que manda es el que más puede.
 3595 |   select rol into v_rol
 3596 |     from public.matriculas
 3597 |    where correo = v_correo and institucion_id = v_inst
 3598 |    order by case rol when 'administrador' then 3 when 'profesor' then 2 else 1 end desc
 3599 |    limit 1;
 3600 | 
 3601 |   if v_rol is null then return 0; end if;
 3602 | 
 3603 |   v_cabe := public.cupo_disponible(v_inst, p_persona);
 3604 | 
 3605 |   update public.perfiles
 3606 |      set -- Quien ya es administración no se baja de rol por una planilla: un
 3607 |          -- archivo mal armado dejaría a la institución sin nadie que pueda
 3608 |          -- arreglarlo.
 3609 |          rol = case when v_actual = 'administrador' then v_actual else v_rol end,
 3610 |          institucion_id = v_inst,
 3611 |          -- El plan sale del contrato, y solo si queda cupo. Sin cupo la
 3612 |          -- persona entra igual, con sus ramos, en el plan gratis: media
 3613 |          -- matrícula es mejor que un error que nadie ve. Quien ya paga por
 3614 |          -- Google Play se queda con lo suyo y no gasta un cupo.
 3615 |          plan = case
 3616 |                   when v_plan = 'personal' then v_plan
 3617 |                   when v_cabe              then 'institucion'
 3618 |                   else v_plan
 3619 |                 end
 3620 |    where id = p_persona;
 3621 | 
 3622 |   -- Los ramos en que queda inscrita.
 3623 |   insert into public.inscripciones (estudiante_id, asignatura_id)
 3624 |   select p_persona, a.id
 3625 |     from public.matriculas m
 3626 |     join public.asignaturas a
 3627 |       on a.codigo = upper(m.codigo) and a.creador_id is null
 3628 |      and a.institucion_id = v_inst
 3629 |    where m.correo = v_correo and m.institucion_id = v_inst
 3630 |      and m.rol = 'estudiante' and m.codigo is not null
 3631 |   on conflict do nothing;
 3632 | 
 3633 |   get diagnostics v_hechas = row_count;
 3634 | 
 3635 |   -- Y los que dicta.
 3636 |   insert into public.dictados (docente_id, asignatura_id, papel)
 3637 |   select p_persona, a.id, coalesce(m.papel, 'profesor')
 3638 |     from public.matriculas m
 3639 |     join public.asignaturas a
 3640 |       on a.codigo = upper(m.codigo) and a.creador_id is null
 3641 |      and a.institucion_id = v_inst
 3642 |    where m.correo = v_correo and m.institucion_id = v_inst
 3643 |      and m.rol = 'profesor' and m.codigo is not null
 3644 |   on conflict do nothing;
 3645 | 
 3646 |   update public.matriculas
 3647 |      set aplicada_en = now()
 3648 |    where correo = v_correo and institucion_id = v_inst and aplicada_en is null;
 3649 | 
 3650 |   return v_hechas;
 3651 | end;
 3652 | $$;
